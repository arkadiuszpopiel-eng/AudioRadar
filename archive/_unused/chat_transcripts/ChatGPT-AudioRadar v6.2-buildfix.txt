import math

class RadarWidget(QtWidgets.QWidget):
    def __init__(self):
        super().__init__()
        self._angle = 0.0
        self._level = 0.0
        self._timer = QtCore.QTimer(self)
        self._timer.setInterval(33)  # ~30 FPS
        self._timer.timeout.connect(self._tick)
        self._timer.start()

    def setLevel(self, lvl: float):
        if lvl is None:
            lvl = 0.0
        # wzmocnienie cichych sygnałów, żeby blip był widoczny
        lvl = max(0.0, min(1.0, float(lvl)))
        self._level = math.sqrt(lvl)

    def _tick(self):
        self._angle = (self._angle + 2.0) % 360.0
        self.update()

    def sizeHint(self):
        return QtCore.QSize(360, 360)

    def paintEvent(self, ev):
        p = QtGui.QPainter(self)
        p.setRenderHint(QtGui.QPainter.Antialiasing)

        r = min(self.width(), self.height()) - 10
        cx = self.width() / 2.0
        cy = self.height() / 2.0
        rad = r / 2.0

        # tło
        p.fillRect(self.rect(), QtGui.QColor(10, 15, 20))

        # siatka
        grid_pen = QtGui.QPen(QtGui.QColor(70, 90, 110))
        grid_pen.setWidth(1)
        p.setPen(grid_pen)
        for k in range(1, 5):
            rr = rad * k / 4.0
            p.drawEllipse(QtCore.QPointF(cx, cy), rr, rr)

        # osie
        p.drawLine(QtCore.QPointF(cx - rad, cy), QtCore.QPointF(cx + rad, cy))
        p.drawLine(QtCore.QPointF(cx, cy - rad), QtCore.QPointF(cx, cy + rad))

        # wskazówka
        angle = self._angle
        beam_pen = QtGui.QPen(QtGui.QColor(120, 220, 255))
        beam_pen.setWidth(2)
        p.setPen(beam_pen)
        x2 = cx + rad * math.cos(math.radians(angle))
        y2 = cy - rad * math.sin(math.radians(angle))
        p.drawLine(QtCore.QPointF(cx, cy), QtCore.QPointF(x2, y2))

        # blip – jasność/rozmiar zależne od poziomu

        blip_color = QtGui.QColor(90, 255, 120, int(100 + 155 * self._level))
        p.setBrush(blip_color)
        p.setPen(QtGui.QPen(QtGui.QColor(0, 0, 0, 0)))

        by = cy - (rad * 0.65) * math.sin(math.radians(angle))
        p.drawEllipse(QtCore.QPointF(bx, by), blip_r, blip_r)

class LevelBuffer:
    """Kolejka poziomów z odczytem NAJNOWSZEJ próbki (bez kumulowania opóźnienia)."""
    def __init__(self): self.q = queue.Queue(maxsize=512)
    def push(self, v):
        try: self.q.put_nowait(float(v))
        except queue.Full:
            try: self.q.get_nowait(); self.q.put_nowait(float(v))
            except Exception: pass
    def pop_latest(self):
        last = None
        try:
            while True:
                last = float(self.q.get_nowait())
        except queue.Empty:
            return last

def prefer_device_index(devs):
    """Preferuj What U Hear/Loopback, 48 kHz, więcej kanałów wejścia."""
    best = None; score_best = -1
    for i, n, info in devs:
        name = n or ''
        ch   = int(info.get('maxInputChannels', 0) or 0)
        sr   = int(info.get('defaultSampleRate', 44100) or 44100)
        score = 0

        if sr == 48000: score += 3
        score += min(ch, 2)
        if score > score_best: best = (i, n, info); score_best = score

def _rms_i16(buf):
    if not buf: return 0.0
    x = np.frombuffer(buf, dtype=np.int16)
    if x.size == 0: return 0.0
    return float(np.sqrt(np.mean((x.astype(np.float32)/32768.0)**2)))
